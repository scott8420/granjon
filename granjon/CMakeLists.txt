cmake_minimum_required(VERSION 3.20)
project(granjon VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type" FORCE)   # -O2 -g: fast, still debuggable (s03; was Debug, -O0)
endif()

find_package(PkgConfig REQUIRED)
pkg_check_modules(GTKMM REQUIRED gtkmm-4.0)
find_package(spdlog REQUIRED)
pkg_check_modules(LIBXML2 REQUIRED libxml-2.0)
# The Preview (M9) sets the compiled font with Pango through a private
# fontconfig. Both come with gtkmm (pango-devel / fontconfig-devel): no new
# package, only their own link lines.
pkg_check_modules(PANGOFT2 REQUIRED pangoft2)
pkg_check_modules(FONTCONFIG REQUIRED fontconfig)
# WOFF2 (M10b): Brotli. Fedora brotli-devel, Ubuntu / Debian libbrotli-dev.
# The decoder only for the selftest's round trip.
pkg_check_modules(BROTLIENC REQUIRED libbrotlienc)
pkg_check_modules(BROTLIDEC REQUIRED libbrotlidec)

# ── Bundled Unicode data -- compiled INTO the core ─────────────────────────────
# The trimmed UCD (names, blocks, decompositions, categories, case), the AGL /
# AGLFN and the char sets live in resources/ (regenerated by scripts/
# ucd_trim.py, checked in, so a build needs no network). tools/embed.cpp turns
# them into one C++ source, so the GTK-free core carries them without a file
# on disk or GLib's resource API. Order of the charsets here = no meaning
# (CharSet sorts them smallest first).
add_executable(granjon_embed tools/embed.cpp)
set(GRANJON_DATA_DIR "${CMAKE_CURRENT_SOURCE_DIR}/resources")
set(GRANJON_DATA_KEYS
    ucd/ucd.txt ucd/blocks.txt ucd/glyphlist.txt ucd/aglfn.txt
    charsets/latin-1.txt charsets/gf-latin-core.txt charsets/adobe-latin-2.txt
    charsets/adobe-latin-3.txt charsets/wgl4.txt)
set(GRANJON_DATA_ARGS "")
set(GRANJON_DATA_FILES "")
foreach(k ${GRANJON_DATA_KEYS})
    list(APPEND GRANJON_DATA_ARGS "${k}=${GRANJON_DATA_DIR}/${k}")
    list(APPEND GRANJON_DATA_FILES "${GRANJON_DATA_DIR}/${k}")
endforeach()
set(GRANJON_DATA_CPP "${CMAKE_CURRENT_BINARY_DIR}/granjon-data.cpp")
add_custom_command(
    OUTPUT  "${GRANJON_DATA_CPP}"
    COMMAND granjon_embed "${GRANJON_DATA_CPP}" ${GRANJON_DATA_ARGS}
    DEPENDS granjon_embed ${GRANJON_DATA_FILES}
    COMMENT "Embedding Unicode data (granjon-data.cpp)"
    VERBATIM)

# ── granjon_core -- the GTK-free core (the model side of the one real seam) ──────
# No gtkmm on this library's link line, BY DESIGN: a stray UI include under
# core/ fails to compile, so the seam is enforced by the build graph and the
# core stays headless-testable. (docs/two-layer-seam.md.)
add_library(granjon_core STATIC
    src/core/Recents.cpp
    src/core/Settings.cpp
    src/core/TextMap.cpp
    # The SHORTCUT-REGISTRY paradigm: one list, two consumers (App wires accels
    # from it, ShortcutsDialog renders from it) so a key and its advertised row
    # can't drift. GTK-free, so it lives in core and the selftest guards it
    # headless -- a pure test proves no two chords collide.
    src/core/Shortcuts.cpp
    # The undo journal itself is header-only (core/Undo.hpp). Granjon's store
    # is the font: model/ is the data, command/ the verbs + the verb table the
    # selftest gate walks. All GTK-free. docs/undo.md, ARCHITECTURE §2-3.
    src/geom/CubicSegment.cpp
    src/geom/Edit.cpp
    src/geom/NodeEdit.cpp   # M5f: insert / delete nodes, line <-> curve
    src/model/Glyph.cpp
    src/model/FontRecords.cpp
    src/model/Font.cpp
    src/model/Bounds.cpp
    src/model/Find.cpp
    src/unicode/Blocks.cpp
    # M3: the bundled UCD, AGL glyph names, char sets; data embedded above.
    src/unicode/UcdTable.cpp
    src/unicode/Agl.cpp
    src/unicode/CharSet.cpp
    src/model/Coverage.cpp
    src/model/Composite.cpp
    src/model/Spacing.cpp
    src/model/Checks.cpp
    src/model/Profiles.cpp
    ${GRANJON_DATA_CPP}
    src/command/Verbs.cpp
    # UFO 3 source format: plist + GLIF via libxml2 (io/). M2.
    src/io/Plist.cpp
    src/io/Glif.cpp
    src/io/Ufo.cpp
    # M9: the OTF compiler -- sfnt assembly, CFF, GPOS kern, the small tables.
    # OtfData.cpp is generated by scripts/otf_tables.py (spec tables, checked in).
    src/export/Sfnt.cpp
    src/export/OtfData.cpp
    src/export/Cff.cpp
    src/export/Tables.cpp
    src/export/Gpos.cpp
    src/export/Otf.cpp
    # M10a: TrueType -- cu2qu (cubic -> quadratic) and glyf / loca / post 2 / gasp.
    src/export/Quad.cpp
    src/export/Ttf.cpp
    # M9c: the drawing codes (CFF / TrueType / SVG path listings) and Export SVG.
    src/export/Codes.cpp
    src/export/Woff2.cpp    # M10b: WOFF2 (Brotli)
    # M11: import OTF / TTF -- our own reader (CFF Type 2 interpreter, glyf,
    # GPOS kern -> groups), no outside library.
    src/import/CffRead.cpp
    src/import/GlyfRead.cpp
    src/import/GposRead.cpp
    src/import/FontReader.cpp
)
target_include_directories(granjon_core PUBLIC include)
# libxml2 parses plist / GLIF. C, GTK-free: it keeps the core's seam intact.
target_include_directories(granjon_core PRIVATE ${LIBXML2_INCLUDE_DIRS})
target_link_libraries(granjon_core PUBLIC ${LIBXML2_LIBRARIES} ${BROTLIENC_LIBRARIES})
target_include_directories(granjon_core PRIVATE ${BROTLIENC_INCLUDE_DIRS})
target_link_directories(granjon_core PUBLIC ${LIBXML2_LIBRARY_DIRS} ${BROTLIENC_LIBRARY_DIRS})
# Vendored nlohmann/json (header-only, MIT), PRIVATE -- an implementation detail
# of the pump, no consumer inherits the include (CANON: surprise-free build).
target_include_directories(granjon_core PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/third_party)

# ── Headless selftest -- exercises the core with no display ────────────────────
# Cairn's Stones (the journal's own fixture) compiles into the selftest only:
# it proves the journal mechanics; the app's store is model::Font.
add_executable(granjon_selftest src/selftest.cpp src/selftest_export.cpp src/selftest_edit.cpp src/selftest_import.cpp src/core/Stones.cpp)
target_link_libraries(granjon_selftest PRIVATE granjon_core ${BROTLIDEC_LIBRARIES})
target_include_directories(granjon_selftest PRIVATE ${BROTLIDEC_INCLUDE_DIRS})
# selftest.cpp's main() is one long walk; GCC 16 with -g notes "variable
# tracking size limit exceeded" and retries without. Skip the attempt for this
# one file (debug info stays; only the location refinement goes). The real
# fix -- main() split per milestone -- is in the ARC backlog.
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    set_source_files_properties(src/selftest.cpp PROPERTIES COMPILE_OPTIONS "-fno-var-tracking-assignments")
endif()

# ── App resources (gresource) -- the RESOURCE-PIPELINE paradigm ────────────────
# glib-compile-resources embeds resources/granjon.gresource.xml into a C source;
# App::on_activate registers it and points the icon theme at the bundle so
# set_icon_name("granjon-logo-symbolic") resolves out of the binary. The generated
# .c compiles as C (project enables C above) so granjon_resources_get_resource()
# has C linkage, matching the extern "C" prototype in App.cpp. Adding an icon:
# drop the .svg in resources/icons/scalable/apps/, list it in the .xml, rebuild.
# (docs/resources-and-dialog-lifetime.md)
find_program(GLIB_COMPILE_RESOURCES glib-compile-resources REQUIRED)
set(GRANJON_GRESOURCE_XML "${CMAKE_CURRENT_SOURCE_DIR}/resources/granjon.gresource.xml")
set(GRANJON_RESOURCES_C   "${CMAKE_CURRENT_BINARY_DIR}/granjon-resources.c")
file(GLOB GRANJON_RESOURCE_FILES
     "${CMAKE_CURRENT_SOURCE_DIR}/resources/icons/scalable/apps/*.svg"
     "${CMAKE_CURRENT_SOURCE_DIR}/resources/*.css")
add_custom_command(
    OUTPUT  "${GRANJON_RESOURCES_C}"
    COMMAND "${GLIB_COMPILE_RESOURCES}"
            --target "${GRANJON_RESOURCES_C}"
            --generate-source
            --c-name granjon_resources
            granjon.gresource.xml
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/resources"
    DEPENDS "${GRANJON_GRESOURCE_XML}" ${GRANJON_RESOURCE_FILES}
    COMMENT "Compiling gresource bundle (granjon-resources.c)"
    VERBATIM)

# ── granjon -- the GTK app ───────────────────────────────────────────────────────
# The Shell is a SPLIT CLASS across TUs by category (see the index in
# include/Shell.hpp); all parts compile into the one granjon::Shell.
add_executable(granjon
    ${GRANJON_RESOURCES_C}
    src/main.cpp
    src/App.cpp
    # Follows the desktop's light/dark preference (GTK4 will not do it for us
    # and we do not link libadwaita). Needs no new dependency: the D-Bus and
    # GSettings calls are GIO, which gtkmm-4.0 already brings in.
    src/Appearance.cpp
    # The house stylesheet: tokens + the Mac/web look (resources/granjon.css),
    # then the user's ~/.config/granjon/styles.css on top.
    src/Style.cpp
    # The panels and views. GTK side of the seam; they read the model and
    # write only through command:: verbs.
    src/views/GlyphPaint.cpp
    src/views/Navigator.cpp
    src/views/GlyphSearch.cpp
    src/views/Inspector.cpp
    src/views/OverviewGrid.cpp
    src/views/GlyphCanvas.cpp
    src/views/SpacingStrip.cpp
    src/views/ReportList.cpp
    src/views/PreviewPane.cpp
    src/Shell.cpp
    src/Shell_zones.cpp
    src/Shell_bindings.cpp
    src/Shell_handlers.cpp
    src/Shell_helpers.cpp
    src/AboutWindow.cpp
    src/ShortcutsDialog.cpp
    src/PreferencesWindow.cpp
    src/Registry.cpp
    src/Log.cpp
)
target_include_directories(granjon PRIVATE include ${GTKMM_INCLUDE_DIRS} ${PANGOFT2_INCLUDE_DIRS} ${FONTCONFIG_INCLUDE_DIRS})
target_link_libraries(granjon     PRIVATE ${GTKMM_LIBRARIES} ${PANGOFT2_LIBRARIES} ${FONTCONFIG_LIBRARIES} granjon_core spdlog::spdlog)
target_link_directories(granjon   PRIVATE ${GTKMM_LIBRARY_DIRS} ${PANGOFT2_LIBRARY_DIRS} ${FONTCONFIG_LIBRARY_DIRS})
target_compile_options(granjon    PRIVATE ${GTKMM_CFLAGS_OTHER})
